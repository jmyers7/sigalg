    Home

    Named after a \(\sigma\)-algebra, a foundational type of object in measure theory, SigAlg is a Python library for computational measure-theoretic probability theory, stochastic calculus, and quantitative finance.

    Installation

    Open a virtual environment running Python 3.13 or 3.14 (3.15 has not been tested yet) and install SigAlg using either uv or pip:

    uv add sigalg

    pip install sigalg

    """Demonstrate Itô's lemma with SigAlg."""
# (1)!

import matplotlib.pyplot as plt
import sigalg as sa

sa.style.set_style("black")

mu = 0.7
sigma = 0.5
T = sa.Time.continuous(start=0, stop=2, dt=0.001)

X = sa.BrownianMotion.generate(
    drift=mu,
    diffusion=sigma,
    initial_state=1,
    n_trajectories=5,
    index=T,
    random_state=42,
)

t = sa.StochasticProcess.from_time(process=X)  # (2)!
W = X.driver

f_t = X**2  # (3)!
f_x = 2 * t * X
f_tt = 0.0
f_xx = 2 * t

ito_integral = sa.ProcessTransforms.ito_integral

increment = ito_integral(integrator=t * X**2)  # (4)!
int_dt = ito_integral(f_t + mu * f_x + (1 / 2) * sigma**2 * f_xx, t)
int_dw = ito_integral(sigma * f_x, W)

fig, axes = plt.subplots(  # (5)!
    ncols=2,
    sharex=True,
    sharey=True,
    figsize=(10, 4),
)

increment.plot_trajectories(
    axes[0], colors=[sa.style.purple, sa.style.teal], plot_kwargs={"linewidth": 1.5}
)
(int_dt + int_dw).plot_trajectories(
    axes[1], colors=[sa.style.purple, sa.style.teal], plot_kwargs={"linewidth": 1.5}
)

axes[0].set_title(r"$f(t, X_t) - f(0, X_0)$")
axes[1].set_title(
    r"$\int_0^t \left( f_t(u, X_u)+\mu f_x(u,X_u) + \frac{1}{2}\sigma^2 f_{xx}(u,X_u) \right) \, du + \int_0^t \sigma f_x(u, X_u) \, dW_u$"
)
axes[0].set_xlabel(r"$t$")
axes[1].set_xlabel(r"$t$")

fig.suptitle(r"Itô's lemma for $f(t,X_t) = tX_t^2$ and $dX_t = 0.7\, dt + 0.5 \, dW_t$")
plt.tight_layout()
plt.show()
    1. Let \(f(t,x)\) be a twice-differentiable function and \(X_t\) a solution to the stochastic differential equation (SDE) \(dX_t = \mu \, dt + \sigma \, dW_t\) where \(\mu\) and \(\sigma>0\) are constants and \(W_t\) is a standard Brownian motion. One simple version of Itô's lemma states that the stochastic process \(f(t, X_t)\) is a solution of the SDE \(df = \left( f_t + \mu f_x + \frac{\sigma^2}{2} f_{xx}\right)\, dt + \sigma f_x \, dW_t\). (For brevity we omit the arguments of all functions in the SDE.)
    2. Get the "time" stochastic process \(t\) and the Brownian motion \(W_t\) that drives \(X_t\).
    3. The function in the demonstration is \(f(t,x) = tx^2\). Compute and set the partial derivatives of this function.
    4. Integrate the differentials in the SDE in the statement of Itô's lemma.
    5. Plot the integrals on the two sides of Itô's lemma to check for their (approximate) equality.
    """Demonstrate the Fundamental Theorem of Calculus with Itô's correction."""
# (1)!

import numpy as np
import sigalg as sa

T = sa.Time.continuous(start=0, stop=4, dt=0.001)
W = sa.BrownianMotion.generate(
    n_trajectories=5,
    index=T,
    name="W",
    random_state=42,
)

ito_integral = sa.ProcessTransforms.ito_integral
quad_var = sa.ProcessTransforms.quad_var


def f(X: sa.StochasticProcess) -> sa.StochasticProcess:
    return X**2


def f_prime(X: sa.StochasticProcess) -> sa.StochasticProcess:
    return 2 * X


increment = ito_integral(integrator=f(W))
classical_term = ito_integral(f_prime(W), W)
ito_correction = ito_integral(integrator=quad_var(W))

np.allclose(increment, classical_term + ito_correction)
    1. The Fundamental Theorem of Calculus with Itô's correction states that \(f(X_t) - f(X_0) = \int_0^t f'(X_u) \, dX_u + \frac{1}{2} \int_0^t f''(X_u) \, d[X]_u\), where \(X_t\) is a "nice" stochastic process (like a diffusion), \(f\) is a twice-differentiable function, and \([X]_t\) is the quadratic variation of \(X_t\). The first term on the right of this equation is the classical term, while the second is Itô's correction.
    """Demonstrate the informal differential 'multiplication' rules in stochastic calculus."""
# (1)!

import matplotlib.pyplot as plt
import numpy as np
import sigalg as sa

sa.style.set_style()
rng = np.random.default_rng(42)

fig, axes = plt.subplots(ncols=3, sharex=True, sharey=True, figsize=(8.5, 3))

quad_var = sa.ProcessTransforms.quad_var
cross_var = sa.ProcessTransforms.cross_var

T = sa.Time.continuous(start=0, stop=1, dt=0.001)
W = sa.BrownianMotion.generate(n_trajectories=5, index=T, name="W", random_state=rng)
t = sa.StochasticProcess.from_time(process=W)

quad_var(W).plot_trajectories(axes[0], colors=[sa.style.purple])
cross_var(t, W).plot_trajectories(axes[1], colors=[sa.style.teal])
quad_var(t).plot_trajectories(axes[2], colors=[sa.style.blue])

axes[0].set_title(r"$[W]_t = t \quad \Rightarrow \quad dW_t^2=0$")
axes[1].set_title(r"$[t,W]_t = 0 \quad \Rightarrow \quad dt \, dW_t=0$")
axes[2].set_title(r"$[t]_t= 0 \quad \Rightarrow \quad dt^2=0$")

fig.suptitle(
    r"Quadratic/cross variations of a Brownian motion $W_t$ and a 'time' process $t$"
)
plt.tight_layout()
plt.show()
    1. In stochastic calculus, the informal differential "multiplication" rules \(dW_t^2=0\), \(dt \, dW_t=0\), and \(dt^2=0\) are used constantly, where \(W_t\) is a standard Brownian motion. These are formal equations with no intrinsic meaning that stand for precise equalities regarding quadratic and cross variations. This snippet demonstrates that these latter equalities hold (approximately, since we're working in the discrete setting.)
    """Demonstrate Black-Scholes pricing and hedging with SigAlg."""

import matplotlib.pyplot as plt
import sigalg as sa

sa.style.set_style("black")

S_0, r, sigma = 50, 0.01, 0.1
T = sa.Time.continuous(start=0, stop=5, dt=0.1)

S = sa.BlackScholes.generate(
    initial_price=S_0,
    risk_free_rate=r,
    volatility=sigma,
    n_trajectories=10,
    index=T,
    random_state=42,
)

Phi = sa.Claim.european_option(strike=52, model=S, type="put")
X = Phi.price_process()  # (1)!
delta = Phi.delta()  # (2)!

_, axes = plt.subplots(ncols=2, sharex=True, figsize=(10, 4))

S.plot_trajectories(axes[0], colors=[sa.style.purple], label="$S_t$")
X.plot_trajectories(axes[0], colors=[sa.style.teal], label="$X_t$")
delta.plot_trajectories(axes[1], colors=[sa.style.blue, sa.style.lime])

axes[0].set_title(r"Prices $S_t$ and $X_t$ of underlying asset and claim")
axes[0].legend()
axes[0].set_xlabel("$t$")
axes[1].set_title(r"$\Delta$-hedging process")
axes[1].set_xlabel("$t$")

plt.tight_layout()
plt.show()
    1. Compute the price process of the European put.
    2. Compute the \(\Delta\)-hedging process.
    """Demo risk-neutral pricing of a European call option with binomial trees."""

import numpy as np
import sigalg as sa

S_0, r, u, d, T = 100, 0.05, 1.1, 0.9, 3

S = sa.BinomialPricingModel.generate(
    initial_price=S_0,
    up_factor=u,
    down_factor=d,
    risk_free_rate=r,
    length=T,
    mode="enum",
)

K = 100
euro_call = sa.Claim.european_option(model=S, strike=K, type="call", name="euro_call")

Delta, B, V = S.replicating_portfolio(claim=euro_call)
X = euro_call.price_process()

X.last_rv == V.last_rv  # (1)!
all(np.allclose(V[t + 1], (1 + r) * B[t] + S[t + 1] * Delta[t]) for t in range(T))

Q = S.emm()  # (2)!
X.discount(r).is_martingale(prob_measure=Q)
    1. The value process \(V_t\) of the replicating portfolio must have two properties: First, at maturity \(T\) the value \(V_T\) must equal the payoff of the option. Second, the portfolio must be self-financing, in the sense that \(V_{t+1} = (1+r) B_t + S_{t+1} \Delta_t\).
    2. The no-arbitrage condition guarnatees the existence of an equivalent martingale measure \(Q\) (also called a risk-neutral measure) such that the discounted price process of the option is a martingale.
    """Demo risk-neutral pricing of an American put option with trinomial trees."""

import numpy as np
import sigalg as sa

S_0, u, m, d, r, T = 100, 1.1, 1.0, 0.9, 0.05, 4

S = sa.TrinomialPricingModel.generate(
    initial_price=S_0,
    up_factor=u,
    middle_factor=m,
    down_factor=d,
    risk_free_rate=r,
    mode="enum",
    length=T,
)


K = 90
american_put = sa.Claim.american_option(
    strike=K,
    model=S,
    type="put",
    name="american_put",
)

theta = 0.5  # (1)!
Q = S.emm(theta=theta)
S.discount(r).is_martingale(prob_measure=Q)

X = american_put.price_process(theta=theta)  # (2)!
X.discount(r).is_martingale(prob_measure=Q, kind="super")

tau = american_put.exercise_time(theta=theta)  # (3)!

price = X[0]  # (4)!
discounted_expectation = american_put.discount(r).at(tau).expectation(measure=Q)
np.allclose(price, discounted_expectation)
    1. Trinomial pricing models are examples of incomplete markets, which means that multiple equivalent martingale measures (EMMs) exist. In SigAlg, the the EMMs are parametrized by a real number \(\theta \in (0,1)\). The discounted price process of the asset is a martingale with respect to every EMM \(Q\).
    2. American option contracts allow early exercise, so their risk-neutral valuation turns into an optimal stopping problem. The price process of an American option with respect to an EMM is not a martingale, but rather a supermartingale.
    3. SigAlg can compute the optimal stopping time \(\tau\) for an early-exercise option.
    4. If we evaluate the discounted price process \((1+r)^{-t}X_t\) of the American option at the optimal stopping time \(\tau\), we get a random variable \((1+r)^{-\tau}X_\tau\) whose expectation is the initial price of the option.
    """Demo a Markov chain in SigAlg."""

from math import prod

import numpy as np
import sigalg as sa

K = [[0.4, 0.6], [0.7, 0.3]]
pi = [0.2, 0.8]
T = 10

X = sa.MarkovChain.generate(kernel=K, initial_distribution=pi, length=T, mode="enum")

P = X.measure  # (1)!
np.allclose(
    P >> X, (P >> X[0]) * prod(P.conditional(X[t]) >> X[t + 1] for t in range(T))
)
    1. The joint distribution \(P\) of a Markov chain factors as \(P(X_0=x_0,\ldots,X_T=x_T) = P(X_0=x_0) \prod_{k=0}^{T-1} P(X_{k+1}=x_{k+1} \mid X_{k} = x_{k})\).
    """Demo polynomial regression as orthogonal projection in an L^2-space."""

import matplotlib.pyplot as plt
import numpy as np
import sigalg as sa

sa.style.set_style()

n = 20
Omega = sa.SampleSpace.from_sequence(size=20)
P = sa.ProbabilityMeasure.uniform(Omega, name="P")
H = sa.L2(Omega, measure=P)  # (1)!

X = sa.RandomVariable(  # (2)!
    domain=Omega,
    measure=P,
    mapping=np.linspace(-2, 2, n),
)
epsilon = sa.RandomVariable.from_rand(
    domain=Omega,
    measure=P,
    distribution="normal",
    name="epsilon",
    random_state=42,
)
Y = (2 + 3 * X - X**2 + epsilon).with_name("Y")

one = sa.RandomVariable.from_constant(  # (3)!
    domain=Omega,
    measure=P,
    constant=1,
    name="one",
)
polynomial_basis = [one, X, X**2, X**3]
Y_fit, coeffs_poly, dim_poly = H.proj(Y, polynomial_basis)

x_smooth = np.linspace(-2, 2, 100)  # (4)!
y_smooth = (
    coeffs_poly[0]
    + coeffs_poly[1] * x_smooth
    + coeffs_poly[2] * x_smooth**2
    + coeffs_poly[3] * x_smooth**3
)

_, ax = plt.subplots()
ax.scatter(X, Y, color=sa.style.amber, s=50)
ax.plot(x_smooth, y_smooth)
plt.tight_layout()
plt.show()
    1. Define the Hilbert space \(\mathcal{H} = L^2(\Omega, \mathcal{F},P)\).
    2. Define a random variable \(X\) on a sample space of cardinality \(20\), and another random variable \(Y = 2+3X-X^2 + \epsilon\) where \(\epsilon\) is a random noise term.
    3. We will project \(Y\) onto the subspace spanned by \(1\), \(X\), and \(X^2\) in an \(L^2\)-space.
    4. Get the coefficients from the orthogonal projection, and plot the fit.
